<script>
const urlParams = new URLSearchParams(window.location.search);
const token = urlParams.get('token');

// Récupération des fragments stockés
let collectedTokens = JSON.parse(localStorage.getItem('rians_tokens') || '[]');

if (token && !collectedTokens.includes(token)) {
collectedTokens.push(token);
localStorage.setItem('rians_tokens', JSON.stringify(collectedTokens));
// Nettoie l'URL pour la clarté
window.history.replaceState({}, document.title, window.location.pathname);
}

// Mise à jour visuelle des cartes GPS et des fragments
const card1 = document.getElementById('card-1');
const card2 = document.getElementById('card-2');
const card3 = document.getElementById('card-3');

const status1 = document.getElementById('status-1');
const status2 = document.getElementById('status-2');
const status3 = document.getElementById('status-3');

const countDisplay = document.getElementById('signals-count');
const display = document.getElementById('fragments-display');
const puzzleSection = document.getElementById('puzzle-section');

let fragmentsText = [];

if (collectedTokens.includes('balise1')) {
card1.classList.add('found');
status1.innerText = "CAPTURES";
status1.style.color = "#00ff00";
fragmentsText.push("1. L'EAU QUI DORT");
}

if (collectedTokens.includes('balise2')) {
card2.classList.add('found');
status2.innerText = "CAPTURES";
status2.style.color = "#00ff00";
fragmentsText.push("2. NE MEURT JAMAIS");
}

if (collectedTokens.includes('balise3')) {
card3.classList.add('found');
status3.innerText = "CAPTURES";
status3.style.color = "#00ff00";
fragmentsText.push("3. ELLE ATTEND ICI");
}

countDisplay.innerText = `${collectedTokens.length}/3 CAPTURE(S)`;

if (fragmentsText.length > 0) {
display.innerHTML = `<strong>FRAGMENTS DE VOIX DÉCRYPTÉS :</strong><br><br>` + fragmentsText.join('<br>');
} else {
display.innerHTML = "<em>Aucun fragment détecté. Approchez-vous d'une balise NFC et scannez-la avec votre téléphone.</em>";
}

if (collectedTokens.length === 3) {
puzzleSection.style.display = 'block';
}

// Validation de la phrase finale
const validateBtn = document.getElementById('validate-btn');
const sentenceInput = document.getElementById('sentence-input');
const statusMsg = document.getElementById('status-msg');

const PHRASE_SECRETE = "L'EAU QUI DORT NE MEURT JAMAIS ELLE ATTEND ICI";

validateBtn.addEventListener('click', () => {
const userText = sentenceInput.value.trim().toUpperCase().replace(/[']/g, "'").replace(/\s+/g, ' ');

if (userText === PHRASE_SECRETE) {
statusMsg.style.color = "#00ff00";
statusMsg.innerText = "SCEAU VALIDÉ. REDIRECTION VERS LE PONT OUATIER...";
localStorage.removeItem('rians_tokens');
setTimeout(() => {
window.location.href = "acte2.html";
}, 2000);
} else {
statusMsg.style.color = "#ff3333";
statusMsg.innerText = "SERMENT INCORRECT. L'ESPRIT DU PARC REJETTE CE MOT.";
}
});
</script>
